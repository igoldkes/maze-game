(from inside maze_game)

windows:

cargo build --release --target x86_64-pc-windows-msvc

    ...

cp target/x86_64-pc-windows-msvc/release/maze_game.exe ./
tar -c -a -f maze-game-win.zip maze_game.exe assets/*


mac:

cargo build --release --target x86_64-apple-darwin

    ...

cargo build --release --target aarch64-apple-darwin

    ...

lipo -create -output maze_game \
    target/x86_64-apple-darwin/release/maze_game \
    target/aarch64-apple-darwin/release/maze_game
zip -r maze-game-mac.zip maze_game assets
